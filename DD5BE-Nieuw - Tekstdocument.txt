async function login() {
  const pat = prompt("Enter GitHub fine-grained token (repo scope):");
  if (!pat) return;
  token = pat.trim(); sessionStorage.setItem("gh_token", token);
  renderAuthStatus(); alert("Authenticated.");

async function renderAccounts(accs) {
  const panel = el("reposPanel"); panel.innerHTML = "<h3>Accounts</h3>";
  for (const acc of accs) {
    const repos = await listReposForAccount(acc);
    const wrap = document.createElement("div");
    wrap.innerHTML = `<strong>${acc.type}:${acc.name}</strong>`;
    repos.forEach(r => {
      const btn = document.createElement("button");
      btn.textContent = `${r.full_name}`;
      btn.onclick = () => selectRepo({ owner: r.owner.login, name: r.name });
      wrap.appendChild(btn);
    });
    panel.appendChild(wrap);
  }
}

async function listReposForAccount(acc) {
  const base = acc.type === "org"
    ? `https://api.github.com/orgs/${acc.name}/repos?per_page=100&type=all`
    : `https://api.github.com/users/${acc.name}/repos?per_page=100&type=all`;
  const res = await fetch(base, { headers: authHeaders() });
  let repos = await res.json();
  return applyFilters(repos, config.filters || {});
}

function applyFilters(repos, filters) {
  const t = (filters.includeTopics||[]).map(s=>s.toLowerCase());
  const ex = (filters.excludeNames||[]);
  const vis = (filters.visibility||["public","private"]);
  return repos.filter(r => {
    const okTopic = !t.length || (Array.isArray(r.topics) && r.topics.some(x => t.includes(x.toLowerCase())));
    const okName = !ex.some(prefix => r.name.startsWith(prefix));
    const okVis = vis.includes(r.private ? "private" : "public");
    return okTopic && okName && okVis;
  });
}

async function selectRepo(repo) {
  if (!token) return alert("Authenticate first.");
  currentRepo = repo;
  const files = await listFiles(repo);
  renderFiles(files);

async function listFiles(repo) {
  const url = `https://api.github.com/repos/${repo.owner}/${repo.name}/git/trees/HEAD?recursive=1`;
  const res = await fetch(url, { headers: authHeaders() });
  const data = await res.json();
  return (data.tree || []).filter(n => n.type === "blob");
}

function renderFiles(files) {

  });
}

async function openFile(path) {
  currentFile = path;
  const url = `https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/contents/${encodeURIComponent(path)}`;
  const res = await fetch(url, { headers: authHeaders() });
  if (!res.ok) return alert("Failed to fetch file.");
  const data = await res.json();
  currentSha = data.sha;
  el("editor").value = atob(data.content);
  preview();
}

function validate() {
  const content = el("editor").value;
  const issues = [];
  if (currentFile?.endsWith(".md") && !/^---\n[\s\S]*\n---/m.test(content)) issues.push("Missing frontmatter (---).");
  const linkRegex = /

\[(.*?)\]

\((.*?)\)/g; let m;
  while ((m = linkRegex.exec(content))) { if (!/^https?:\/\//.test(m[2])) issues.push(`Non-HTTP link: ${m[2]}`); }
  alert(issues.length ? `Issues:\n- ${issues.join("\n- ")}` : "Validation passed.");
}

function preview() {
  const content = el("editor").value, iframe = el("preview");
  const isHTML = currentFile?.endsWith(".html");
  iframe.srcdoc = isHTML ? content : `<pre style="white-space:pre-wrap;font-family:monospace">${escapeHtml(content)}</pre>`;
}

async function commit() {
  if (!token || !currentRepo || !currentFile) return alert("Login/select repo/open file first.");
  const newContent = el("editor").value;
  const msgTpl = (config.workflows?.commitMessageTemplate || "CMS: {file} — {action}");
  const msg = msgTpl.replace("{file}", currentFile).replace("{action}", "update");
  const body = { message: msg, content: btoa(unescape(encodeURIComponent(newContent))), sha: currentSha };
  const url = `https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/contents/${encodeURIComponent(currentFile)}`;
  const putRes = await fetch(url, { method: "PUT", headers: authHeaders(true), body: JSON.stringify(body) });
  if (!putRes.ok) return alert("Commit failed.");
  const resData = await putRes.json();
  currentSha = resData.content?.sha || currentSha;
  alert("Committed.");
}

async function openPR() {
  if (!token || !currentRepo) return alert("Login/select repo first.");
  const base = prompt("Base branch (default: main):", "main") || "main";
  const head = prompt("New branch name:", `cms-${Date.now()}`); if (!head) return;
  const baseRefUrl = `https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/git/ref/heads/${base}`;
  const baseRefRes = await fetch(baseRefUrl, { headers: authHeaders() }); if (!baseRefRes.ok) return alert("Failed to read base ref.");
  const baseRef = await baseRefRes.json();
  const createRefUrl = `https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/git/refs`;
  const refRes = await fetch(createRefUrl, { method: "POST", headers: authHeaders(true), body: JSON.stringify({ ref: `refs/heads/${head}`, sha: baseRef.object.sha }) });
  if (!refRes.ok) return alert("Failed to create branch.");
  const prUrl = `https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/pulls`;
  const prRes = await fetch(prUrl, { method: "POST", headers: authHeaders(true), body: JSON.stringify({ title: `CMS update: ${currentFile}`, head, base, body: "Automated via CMS." }) });
  if (!prRes.ok) return alert("PR creation failed.");
  const pr = await prRes.json(); alert(`PR opened: #${pr.number}`);
}

}

  pull_request:

  check:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Link check (basic)
        run: |
          grep -Rho '\(http[s]\?://[^") ]\+\)' -n . | wc -l
      - name: Frontmatter check
        run: |
          if ls **/*.md >/dev/null 2>&1; then
            for f in $(ls **/*.md); do
              head -n 1 "$f" | grep -q '---' || (echo "Missing frontmatter: $f" && exit 1)
            done
          fi

Gateway architecture for Samsung A53 and universal mobile support
You want your Samsung A53 to be your secure gateway and to extend the same principle to all past, present, and future phones—without excluding any device or requiring manufacturers to disclose plans. Below is a complete, implementable blueprint plus runnable stubs that integrate with your existing CMS and add a “virtual room” as an extra-frame without changing your UI/UX/GUI.

Principle and trust model
Identity kernel:

Device ID: A stable identifier with hardware-backed keys where available.

Key material: Generate a public/private key pair; private key stays on the device.

Gateway ID: Format like T3-GATEWAY:<KeyFingerprint>@<Domain>.

Capability manifest:

Channels: IP/HTTPS, BLE, NFC, USB, Wi‑Fi Direct, SMS/USSD, Voice/DTMF.

Crypto: Asymmetric keys, symmetric session keys, signed challenges.

Connector framework:

Adapters: IP/Web (HTTP, WebSocket, WebRTC), BLE, NFC, USB, SMS, Voice IVR.

Operations: connect, authenticate, publish, subscribe, close; uniform API across channels.

Triangular trust:

Client (computer) ↔ Gateway (phone) ↔ Platform (CMS/Portal).

Mutual auth: Challenge-response signed by device key; platform verifies and issues short-lived session tokens.

Network compliance: Only operate within allowed domains/interfaces; no bypassing local policies.

Device profiles that include all phones
Profile S — Smart devices (Android/iOS/HarmonyOS):

Capabilities: Background services (where allowed), hardware keystore, IP/BLE/NFC/Wi‑Fi Direct.

Form factor: Native app or PWA for gateway; full feature support.

Profile W — Web-capable but limited:

Capabilities: PWA gateway, browser storage for keys; optional WebBluetooth/WebNFC.

Mode: Session-based gateway while browser is open.

Profile L — Feature phones (SMS/USSD/Voice):

Capabilities: Identity via SIM/number; actions via SMS/USSD/DTMF IVR.

Role: Approval second factor, session unlocks, deploy confirmations.

Profile H — Future hardware:

Capabilities: Unknown/new protocols; declare via dynamic profile mapping.

Contract: Open gateway protocol; secure, documented, manufacturer-agnostic.

Samsung A53 gateway app (Android) specifics
Services:

Foreground service: Always-on, starts at boot, maintains secure channel (WSS/WebRTC) to platform.

BLE advertiser: Proximity discovery with rotating identifiers.

mDNS announcer: LAN discovery where permitted.

NFC/USB pairing: One-time hard-pair to bind trust.

Wi‑Fi Direct (optional): Direct session bootstrap when network policy allows.

Security:

Android Keystore: Store private keys; sign challenges locally.

Session tokens: Request short-lived tokens from platform after successful mutual auth.

Policy checks: Device enforces connector use based on site/network rules.

async function listReposForAccount(acc){
  const base = acc.type==="org"
    ? `https://api.github.com/orgs/${acc.name}/repos?per_page=100&type=all`
    : `https://api.github.com/users/${acc.name}/repos?per_page=100&type=all`;
  const res = await fetch(base,{headers:authHeaders()});
  let repos = await res.json();
  return applyFilters(repos, config.filters||{});
}

function applyFilters(repos, filters){
  const t=(filters.includeTopics||[]).map(s=>s.toLowerCase());
  const ex=(filters.excludeNames||[]);
  const vis=(filters.visibility||["public","private"]);
  return repos.filter(r=>{
    const okTopic=!t.length||(Array.isArray(r.topics)&&r.topics.some(x=>t.includes(x.toLowerCase())));
    const okName=!ex.some(prefix=>r.name.startsWith(prefix));
    const okVis=vis.includes(r.private?"private":"public");
    return okTopic && okName && okVis;
  });
}

function validate(){
  const content=el("editor").value;
  const issues=[];
  if(currentFile?.endsWith(".md") && !/^---\n[\s\S]*\n---/m.test(content)) issues.push("Missing frontmatter (---).");
  const linkRegex=/

\[(.*?)\]

\((.*?)\)/g; let m;
  while((m=linkRegex.exec(content))){ if(!/^https?:\/\//.test(m[2])) issues.push(`Non-HTTP link: ${m[2]}`); }
  alert(issues.length?`Issues:\n- ${issues.join("\n- ")}`:"Validation passed.");
}

async function commit(){
  if(!token||!currentRepo||!currentFile) return alert("Login/select repo/open file first.");
  const newContent=el("editor").value;
  const msgTpl=(config.workflows?.commitMessageTemplate || "CMS: {file} — {action}");
  const msg=msgTpl.replace("{file}",currentFile).replace("{action}","update");
  const body={message:msg, content:btoa(unescape(encodeURIComponent(newContent))), sha:currentSha};
  const url=`https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/contents/${encodeURIComponent(currentFile)}`;
  const putRes=await fetch(url,{method:"PUT", headers:authHeaders(true), body:JSON.stringify(body)});
  if(!putRes.ok) return alert("Commit failed.");
  const resData=await putRes.json(); currentSha=resData.content?.sha||currentSha;
  alert("Committed.");
}

async function openPR(){
  if(!token||!currentRepo) return alert("Login/select repo first.");
  const base=prompt("Base branch (default: main):","main")||"main";
  const head=prompt("New branch name:",`cms-${Date.now()}`); if(!head) return;
  const baseRefUrl=`https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/git/ref/heads/${base}`;
  const baseRefRes=await fetch(baseRefUrl,{headers:authHeaders()}); if(!baseRefRes.ok) return alert("Failed to read base ref.");
  const baseRef=await baseRefRes.json();
  const createRefUrl=`https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/git/refs`;
  const refRes=await fetch(createRefUrl,{method:"POST", headers:authHeaders(true), body:JSON.stringify({ref:`refs/heads/${head}`, sha:baseRef.object.sha})});
  if(!refRes.ok) return alert("Failed to create branch.");
  const prUrl=`https://api.github.com/repos/${currentRepo.owner}/${currentRepo.name}/pulls`;
  const prRes=await fetch(prUrl,{method:"POST", headers:authHeaders(true), body:JSON.stringify({title:`CMS update: ${currentFile}`, head, base, body:"Automated via CMS."})});
  if(!prRes.ok) return alert("PR creation failed.");
  const pr=await prRes.json(); alert(`PR opened: #${pr.number}`);
}

async function listReposForAccount(acc){
  const base = acc.type==="org"
    ? `https://api.github.com/orgs/${acc.name}/repos?per_page=100&type=all`
    : `https://api.github.com/users/${acc.name}/repos?per_page=100&type=all`;
  const res = await fetch(base,{headers:authHeaders()});
  let repos = await res.json();
  return applyFilters(repos, config.filters||{});

function applyFilters(repos, filters){
  const t=(filters.includeTopics||[]).map(s=>s.toLowerCase());
  const ex=(filters.excludeNames||[]);
  const vis=(filters.visibility||["public","private"]);
  return repos.filter(r=>{
    const okTopic=!t.length||(Array.isArray(r.topics)&&r.topics.some(x=>t.includes(x.toLowerCase())));
    const okName=!ex.some(prefix=>r.name.startsWith(prefix));
    const okVis=vis.includes(r.private?"private":"public");
    return okTopic && okName && okVis;
  });
}

// Accessibility: focus trap
let lastFocused = null;
function trapFocus(e) {
  const focusable = overlay.querySelectorAll('a, button, input, [tabindex]:not([tabindex="-1"])');
  if (!focusable.length) return;
  const first = focusable[0], last = focusable[focusable.length - 1];
  if (e.key === "Tab") {
    if (e.shiftKey && document.activeElement === first) { e.preventDefault(); last.focus(); }
    else if (!e.shiftKey && document.activeElement === last) { e.preventDefault(); first.focus(); }
  }
}

function showIntroIfNeeded() {
  try {
    const stored = localStorage.getItem("intro_skip");
    const isSkip = stored === "true";
    // Always announce at first run; allow skip thereafter
    if (!isSkip) {
      overlay.hidden = false;
      lastFocused = document.activeElement;
      overlay.addEventListener("keydown", trapFocus);
      // Move focus to the dialog title
      const title = document.getElementById("introTitle");
      title && title.focus && title.focus();
    }
  } catch {}
}
function closeIntro() {
  overlay.hidden = true;
  overlay.removeEventListener("keydown", trapFocus);
  if (skip?.checked) localStorage.setItem("intro_skip", "true");
  if (lastFocused?.focus) lastFocused.focus();
}
btn?.addEventListener("click", closeIntro);

// Auto-run after DOM is ready
